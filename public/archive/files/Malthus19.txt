import numpy as np
SimulationTime = 1.0
dt = 0.001
GrowthRate = 1.0
Population = 1.0

for Time in np.arange(dt, SimulationTime+dt, dt):
    Population = Population + dt * GrowthRate * Population
    print('Time = ' + str(Time) + '; Population = ' + str(Population) + '.')
