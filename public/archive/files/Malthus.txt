import numpy as np
SimulationTime = 10.0
dt = 0.1
GrowthRate = 0.2
Population = 1.0

for Time in np.arange(dt, SimulationTime, dt):
    Population = Population + dt * GrowthRate * Population
    print('Time = ' + str(Time) + '; Population = ' + str(Population) + '.')
